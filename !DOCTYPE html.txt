<script>
  const PARTICIPANTS = 16;
  const START_AMOUNT = 1000000;
  const LIVING_COST_PER_PERSON = 10000;
  const MAX_ROUNDS = 6;
  const MULTIPLIER = 2;

  let currentRound = 0;
  let currentPool = START_AMOUNT;
  let history = [];

  function formatMoney(num) {
    return Math.round(num).toLocaleString('ko-KR') + '원';
  }

  function updateDisplay() {
    document.getElementById('roundNum').textContent = currentRound;
    document.getElementById('poolAmount').textContent = formatMoney(currentPool);

    if (history.length > 0) {
      document.getElementById('historyTable').style.display = 'table';
      const tbody = document.getElementById('historyBody');
      tbody.innerHTML = '';
      history.forEach(h => {
        const tr = document.createElement('tr');
        tr.innerHTML = `
          <td>${h.round}</td>
          <td>${formatMoney(h.afterCost)}</td>
          <td>${formatMoney(h.afterMultiply)}</td>
          <td>${formatMoney(h.perPerson)}</td>
        `;
        tbody.appendChild(tr);
      });
    }

    if (currentRound >= MAX_ROUNDS) {
      document.getElementById('nextBtn').disabled = true;
      document.getElementById('nextBtn').style.opacity = '0.5';
      document.getElementById('nextBtn').textContent = '✅ 최대 라운드 도달';
    }
  }

  function nextRound() {
    if (currentRound >= MAX_ROUNDS) return;

    currentRound++;
    const livingCost = PARTICIPANTS * LIVING_COST_PER_PERSON;
    const afterCost = currentPool - livingCost;
    const afterMultiply = afterCost * MULTIPLIER;
    const perPerson = afterCost / PARTICIPANTS;

    history.push({
      round: currentRound,
      afterCost: afterCost,
      afterMultiply: afterMultiply,
      perPerson: perPerson
    });

    if (currentRound >= MAX_ROUNDS) {
      currentPool = afterCost; // 마지막 라운드는 2배 적용 안하고 바로 분배
      showResult(true);
    } else {
      currentPool = afterMultiply;
      updateDisplay();
    }
  }

  function stopAndDistribute() {
    if (currentRound === 0) {
      alert('아직 라운드를 진행하지 않았습니다. 먼저 라운드를 진행해주세요!');
      return;
    }
    showResult(false);
  }

  function showResult(isMaxRound) {
    document.getElementById('mainScreen').classList.add('hidden');
    document.getElementById('resultScreen').classList.add('active');

    const finalPool = currentPool;
    const perPerson = finalPool / PARTICIPANTS;

    document.getElementById('finalAmount').textContent = formatMoney(finalPool);
    document.getElementById('finalRound').textContent = currentRound;
    document.getElementById('perPersonAmount').textContent = formatMoney(perPerson);
    document.getElementById('compareResult').textContent = formatMoney(perPerson);

    const multiple = (perPerson / 100000).toFixed(1);
    
    let title = isMaxRound 
      ? '🎉 6라운드 완주! 최종 결과' 
      : `🤝 ${currentRound}라운드에서 분배 결정!`;
    document.getElementById('resultTitle').textContent = title;

    let message = '';
    if (perPerson >= 1000000) {
      message = `🎊 축하합니다! 협력을 통해 원금(100만원)의 <b>${multiple}배</b>인 <b>${formatMoney(perPerson)}</b>를 각자 받게 되었습니다!<br><br>
      만약 1라운드에서 성급하게 혼자 100,000원(1/16)만 가져갔다면 이렇게 큰 금액을 받지 못했을 것입니다.<br><br>
      <b>💡 교훈: 협력하고 기다리면 모두가 더 큰 이익을 얻을 수 있습니다.</b>`;
    } else {
      message = `현재까지는 1인당 <b>${formatMoney(perPerson)}</b>를 받게 됩니다.<br><br>
      더 오래 기다렸다면 (최대 6라운드까지) 훨씬 더 큰 금액을 받을 수 있었습니다!<br><br>
      <b>💡 교훈: 조급하게 멈추기보다 끝까지 기다리는 것이 더 유리할 수 있습니다.</b>`;
    }
    document.getElementById('messageBox').innerHTML = message;
  }

  function resetGame() {
    currentRound = 0;
    currentPool = START_AMOUNT;
    history = [];
    document.getElementById('mainScreen').classList.remove('hidden');
    document.getElementById('resultScreen').classList.remove('active');
    document.getElementById('nextBtn').disabled = false;
    document.getElementById('nextBtn').style.opacity = '1';
    document.getElementById('nextBtn').textContent = '▶ 다음 라운드 진행';
    document.getElementById('historyTable').style.display = 'none';
    updateDisplay();
  }

  updateDisplay();
</script>
